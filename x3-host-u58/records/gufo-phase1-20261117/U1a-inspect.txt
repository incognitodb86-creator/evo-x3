# Start llama-swap on loopback. Canonical config keeps ${env.LLM_KEY_*} placeholders.

set -euo pipefail
CONFIG=/etc/evo-x3/llama-swap.yaml
ENVFILE=/etc/evo-x3/llm.env
LISTEN="${LISTEN_ADDR:-127.0.0.1:8080}"
BIN=/usr/local/bin/llama-swap

if [[ -f "$ENVFILE" ]]; then
  set -a
  # shellcheck disable=SC1090
  source "$ENVFILE"
  set +a
fi

keys_present=0
while IFS= read -r name; do
  [[ -n "${!name-}" ]] && keys_present=1
done < <(compgen -v | grep '^LLM_KEY_' || true)

if [[ "$keys_present" -eq 1 ]]; then
  exec "$BIN" -config "$CONFIG" -listen "$LISTEN"
fi
