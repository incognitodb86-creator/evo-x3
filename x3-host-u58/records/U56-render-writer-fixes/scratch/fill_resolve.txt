    62	    prompt_text: str, run_dir: Path, ask: str = "", position: str = "", still_image: Optional[str] = None,
    63	    partner2_still: Optional[str] = None, character: Optional[str] = None, names: Optional[Sequence[str]] = None,
    64	    refmod_stems: Optional[Sequence[str]] = None, template_name: Optional[str] = None, still_mode: str = "guide",
    65	    width: int = 768, height: int = 1344, frames: Optional[int] = None, seed: int = 51083601,
    66	    continuation_prompt: Optional[str] = None, dry_run: bool = True,

    69	    still_image, partner2_still = pixl_abs(still_image), pixl_abs(partner2_still)
    70	    rep: Dict[str, Any] = {"hook": "fill_request", "ask": ask, "position": position, "seed": seed, "frames": frames, "status": "NOT_READY", "submitted": False}
    71

    73	        rep.update(status="NOT_READY", reason=reason)
    74	        (run_dir / "clip_NOT_READY.txt").write_text(f"NOT_READY: {reason}\n", encoding="utf-8")
    75	        (run_dir / "build_report.json").write_text(json.dumps(rep, indent=2), encoding="utf-8"); return dict(rep, run_dir=str(run_dir))
    76	
    77	    raw = (prompt_text or "").strip()
    78	    if not raw: return _fail("empty writer output")
    79	    if (m := br._REFUSAL_RE.search(br._D_RE.sub("", raw))): return _fail(f"refusal phrase: {m.group(0)!r}")
    80	    try:
    81	        br._check_writer_18plus(raw, {"ask": ask})

    83	        return _fail(str(e).removeprefix("NOT_READY: "))
    84	
    85	    stems, id_info = list(refmod_stems or []), {}

    87	        picked = res.resolve_writer_refmods(still_image=still_image, character=character, names=names)
    88	        if (character or names) and picked.get("reason") and not picked.get("stems"):
    89	            return _fail(picked["reason"])

   100	    (run_dir / "prompt.txt").write_text(final_prompt + "\n", encoding="utf-8")
   101	
   102	    c0_in = g["c0"]["inputs"]

   105	        c0_in["length"] = int(br.align_h3_frames(int(frames)))

   107	    if (s_rel := _stage_still(still_image)):
   108	        for nid in ("img_1", "img"):
   109	            if nid in g: g[nid]["inputs"]["image"] = s_rel; break

    50	def validate_biometric_description(bio: Optional[str]) -> str:
    51	    """Validate biometric / wardrobe description against empty and generic-only inputs.
    52	
    53	    Args:
    54	        bio: Input description string.
    55	
    56	    Returns:
    57	        Cleaned, non-empty description string containing concrete biometric/wardrobe details.
    58	
    59	    Raises:
    60	        ValueError: If bio is missing/empty, matches banned generics alone,
    61	            or contains only generic phrasing without concrete details.
    62	    """
    63	    if bio is None:
    64	        raise ValueError("Biometric description / bio is required: cannot be None.")
    65	
    66	    text = str(bio).strip()
    67	    if not text:
    68	        raise ValueError("Biometric description / bio is required: cannot be empty or whitespace.")
    69	
    70	    norm = re.sub(r"[^\w\s]", " ", text.lower())
    71	    norm_words = norm.split()
    72	    norm_clean = " ".join(norm_words)
    73	
    74	    # 1. Exact match to banned generic phrases alone
    75	    for banned in BANNED_GENERICS:
    76	        banned_norm = " ".join(re.sub(r"[^\w\s]", " ", banned.lower()).split())
    77	        if norm_clean == banned_norm:
    78	            raise ValueError(
    79	                f"Biometric description matches banned generic phrase alone: {text!r}. "
    80	                "Must provide concrete biometric/wardrobe details."
    81	            )
    82	
    83	    # 2. Check if residual after removing banned phrases and generic filler words is empty
    84	    residual = norm_clean
    85	    extended_banned = list(BANNED_GENERICS) + [
    86	        "exact likeness and vocal timbre",
    87	        "distinctive facial features and vocal timbre",
    88	        "natural authentic human appearance",
    89	        "authentic human appearance",
    90	        "natural human appearance",
    91	        "natural vocal timbre",
    92	        "vocal timbre",
    93	    ]
    94	    for banned in sorted(extended_banned, key=len, reverse=True):
    95	        banned_norm = " ".join(re.sub(r"[^\w\s]", " ", banned.lower()).split())
    96	        residual = residual.replace(banned_norm, " ")
    97	
    98	    filler_words = {
    99	        "and", "with", "the", "a", "an", "of", "in", "to", "for", "styled", "tone",
   100	        "features", "appearance", "likeness", "facial", "vocal", "timbre", "natural",
   101	        "authentic", "exact", "distinctive", "human", "alone", "only", "just",
   102	    }
   103	    residual_tokens = [w for w in residual.split() if w not in filler_words]
   104	    if not residual_tokens:
   105	        raise ValueError(
   106	            f"Biometric description contains only generic phrasing without concrete details: {text!r}."
   107	        )
   108	
   109	    return text

   112	def load_index_files(index_dir: Path) -> List[Dict[str, Any]]:
   113	    """Parse INDEX.md and INDEX_BAD.md into a list of row dicts.
   114	
   115	    Args:
   116	        index_dir: Directory containing INDEX.md and INDEX_BAD.md.
   117	
   118	    Returns:
   119	        List of character dictionaries with keys:
   120	            - character: character name
   121	            - actor: real actor name
   122	            - franchise: franchise / show
   123	            - clips: list of clip filenames
   124	            - date_added: date string
   125	            - status: "good", "onthefence", or "bad"
   126	    """
   127	    rows: List[Dict[str, Any]] = []
   128	
   129	    index_main = index_dir / "INDEX.md"
   130	    if index_main.is_file():
   131	        current_status = "good"
   132	        for line in index_main.read_text(encoding="utf-8").splitlines():
   133	            line_str = line.strip()

   135	                current_status = "good"

   138	            elif line_str.startswith("|") and not line_str.startswith("| :---") and not line_str.startswith("| Character"):
   139	                parts = [p.strip() for p in line_str.split("|")[1:-1]]
   140	                if len(parts) >= 5:
   141	                    char_name = re.sub(r"^\*+|\*+$", "", parts[0]).strip()
   142	                    actor_name = parts[1].strip()
   143	                    franchise = parts[2].strip()
   144	                    clips = [c.strip() for c in parts[3].split("<br>") if c.strip()]
   145	                    date_added = parts[4].strip()
   146	                    rows.append({
   147	                        "character": char_name,
   148	                        "actor": actor_name,
   149	                        "franchise": franchise,

   239	def validate_reference_images(
   240	    images: Optional[List[str]],
   241	    pools: Optional[Tuple[Path, ...]] = None,
   242	) -> List[str]:
   243	    """U2: --image gate (LANE_IMAGE_RULES.md). Image paths must exist; bare names are looked up in the pools.
   244	
   245	    Accepts a full path, or a sub-path / filename relative to a pool. Rejects > 4 images
   246	    (MAX_REFERENCE_IMAGES), missing files, and non-image extensions.
   247	    build_request.py wires --image as a single frame-0 AddGuide start image (0 or 1 per clip).
   248	    """
   249	    if not images:
   250	        return []
   251	    imgs = [str(x).strip() for x in images if x and str(x).strip()]
   252	    if len(imgs) > MAX_REFERENCE_IMAGES:
   253	        raise ValueError(f"at most {MAX_REFERENCE_IMAGES} --image references per clip (got {len(imgs)})")
   254	    pool_list = tuple(Path(x) for x in (pools if pools is not None else APPROVED_IMAGE_POOLS))
   255	    out: List[str] = []
   256	    for raw in imgs:
   257	        cand = Path(raw).expanduser() if raw.startswith("~") else Path(raw)
   258	        tries = [cand] if cand.is_absolute() or str(cand).startswith("\\\\") else [pl / cand for pl in pool_list]
   259	        found = None
   260	        for t in tries:
   261	            try:
   262	                rp = t.resolve()
   263	            except OSError:
   264	                continue
   265	            # Pool containment dropped (Derrick t2086u): any existing image path is accepted;
   266	            # bare names are still looked up inside the pools for convenience.
   267	            if rp.is_file():
   268	                found = rp
   269	                break
   270	        if found is None:
   271	            raise ValueError(f"--image '{raw}' not found (absolute path, or name inside a pool)")
   272	        if found.suffix.lower() not in REFERENCE_IMAGE_EXTS:
   273	            raise ValueError(f"--image '{raw}' must be one of {', '.join(REFERENCE_IMAGE_EXTS)}")
   274	        out.append(str(found))
   275	    return out

   278	def resolve_identities(
   279	    request: Dict[str, Any],
   280	    index_dir: Path,
   281	    refmods_dir: Path,
   282	    approved_stills_dir: Optional[Path] = None,
   283	    aliases_path: Optional[Path] = None,
   284	) -> Dict[str, Any]:
   285	    """Resolve normalized request against INDEX and RefMods.
   286	
   287	    Args:
   288	        request: Normalized request dict from request.py.
   289	        index_dir: Directory containing INDEX.md and INDEX_BAD.md.
   290	        refmods_dir: Directory containing *.safetensors RefMod files.
   291	        approved_stills_dir: Approved pool directory for stills gate.
   292	        aliases_path: Path to aliases.json.
   293	
   294	    Returns:
   295	        Resolved dictionary with mode, INDEX rows, RefMod slots (1..8), description.
   296	    """
   297	    index_rows = load_index_files(index_dir)
   298	    aliases = load_aliases(aliases_path)
   299	
   300	    # 1. Validate stills (U3)
   301	    validated_stills = validate_stills(request.get("stills", []), approved_stills_dir)
   302	
   303	    # 2. Gather candidate identities
   304	    names = request.get("names", [])
   305	    character_input = request.get("character")
   306	    actor_input = request.get("actor")
   307	
   308	    candidates: List[str] = []
   309	    for n in names:
   310	        if n not in candidates:
   311	            candidates.append(n)
   312	    if character_input and character_input not in candidates:
   313	        candidates.append(character_input)
   314	    if actor_input and actor_input not in candidates:
   315	        candidates.append(actor_input)
   316	
   317	    # Expand aliases (KD2 stacking)
   318	    expanded_candidates: List[Tuple[str, str]] = []  # (name, role: "character" | "actor" | "query")
   319	    for c in candidates:
   320	        norm_c = _normalize_key(c)
   321	        expanded_candidates.append((c, "character" if c == character_input else "query"))
   322	        if norm_c in aliases:
   323	            resolved_actor = aliases[norm_c]
   324	            if resolved_actor not in [ec[0] for ec in expanded_candidates]:
   325	                expanded_candidates.append((resolved_actor, "actor"))
   326	
   327	    # 3. Match against INDEX rows
   328	    matched_index_rows: List[Dict[str, Any]] = []
   329	    for c_name, role in expanded_candidates:
   330	        norm_c = _normalize_key(c_name)
   331	        for r in index_rows:
   332	            norm_char = _normalize_key(r["character"])
   333	            norm_act = _normalize_key(r["actor"])
   334	            if norm_c == norm_char or norm_c == norm_act or norm_c in norm_char or norm_c in norm_act:
   335	                if r not in matched_index_rows:
   336	                    matched_index_rows.append(r)
   337	                    # If matched character, also add its actor if not present
   338	                    if r["actor"] and r["actor"] != r["character"]:
   339	                        actor_cand = (r["actor"], "actor")
   340	                        if actor_cand not in expanded_candidates:
   341	                            expanded_candidates.append(actor_cand)
   342	
   343	    # Determine overall INDEX status
   344	    # Priority: if a character is explicitly requested and matched in index, its status governs.
   345	    # Otherwise: good > onthefence > bad > none
   346	    status_priority = {"good": 3, "onthefence": 2, "bad": 1}
   347	    best_status = "none"
   348	    if matched_index_rows:
   349	        if character_input:
   350	            char_norm = _normalize_key(character_input)
   351	            char_matches = [r for r in matched_index_rows if char_norm == _normalize_key(r["character"])]
   352	            if char_matches:
   353	                best_status = char_matches[0]["status"]
   354	        if best_status == "none":
   355	            statuses = [r["status"] for r in matched_index_rows]
   356	            sorted_statuses = sorted(statuses, key=lambda s: status_priority.get(s, 0), reverse=True)
   357	            best_status = sorted_statuses[0]
   358	
   359	    # Mode determination (KD9, KD13, KD15)
   360	    if best_status == "good":
   361	        mode = "native_face_voice"
   362	    elif best_status in ("bad", "onthefence"):
   363	        mode = "native_voice_refmod_look"
   364	    else:
   365	        mode = "native_voice_refmod_look"  # Not in INDEX treated as bad per KD15
   366	
   367	    # 4. Find RefMod files
   368	    matched_refmods: List[Tuple[Path, str, str]] = []  # (path, role, status)
   369	    for c_name, role in expanded_candidates:
   370	        norm_c = _normalize_key(c_name)
   371	        # Find index status for this specific candidate if any
   372	        c_status = "bad"
   373	        for r in matched_index_rows:
   374	            if norm_c == _normalize_key(r["character"]) or norm_c == _normalize_key(r["actor"]):
   375	                c_status = r["status"]
   376	                break
   377	
   378	        files = find_refmods_for_name(norm_c, refmods_dir)
   379	        for f in files:
   380	            if not any(f == m[0] for m in matched_refmods):
   381	                matched_refmods.append((f, role, c_status))
   382	
   383	    # 5. Over 8 RefMods truncation (KD14):
   384	    # Keep character matches first, then fill with highest INDEX status (good > fence > bad), drop rest
   385	    def _refmod_sort_key(item: Tuple[Path, str, str]) -> Tuple[int, int]:
   386	        role_weight = 1 if item[1] == "character" else 0
   387	        stat_weight = status_priority.get(item[2], 0)
   388	        return (role_weight, stat_weight)
   389	
   390	    sorted_refmods = sorted(matched_refmods, key=_refmod_sort_key, reverse=True)
   391	    kept_refmods = sorted_refmods[:8]
   392	    dropped_refmods = sorted_refmods[8:]
   393	
   394	    # Build 8 fixed loader slots (F3)
   395	    slots: Dict[str, Dict[str, Any]] = {}
   396	    for i in range(1, 9):
   397	        slot_key = f"mod_{i}"
   398	        if i - 1 < len(kept_refmods):
   399	            rf_path, _, _ = kept_refmods[i - 1]
   400	            strength = KD17_REFMOD_STRENGTHS[min(i - 1, len(KD17_REFMOD_STRENGTHS) - 1)]
   401	            slots[slot_key] = {"path": str(rf_path), "strength": strength, "copies": 1}
   402	        else:
   403	            slots[slot_key] = {"path": "(none)", "strength": 1.0, "copies": 1}
   404	
   405	    # 6. Build written identity description (KD3)
   406	    raw_bio = request.get("description") or request.get("bio")
   407	    valid_bio = validate_biometric_description(raw_bio)
   408	
   409	    primary_name = candidates[0] if candidates else "subject"
   410	    # MA-6: strip trailing period on bio before appending ", styled in ... tone."
   411	    valid_bio = valid_bio.rstrip().rstrip(".")
   412	    # Same-name celebrity: (actor X) not (character X, actor X)
   413	    same_name = (
   414	        bool(character_input)
   415	        and bool(actor_input)
   416	        and _normalize_key(character_input) == _normalize_key(actor_input)
   417	        and _normalize_key(character_input) == _normalize_key(primary_name)
   418	    )
   419	    if same_name:
   420	        parts = [f"actor {actor_input}"]
   421	    else:
   422	        char_part = f"character {character_input}" if character_input else ""
   423	        actor_part = f"actor {actor_input}" if actor_input else ""
   424	        parts = [p for p in [char_part, actor_part] if p]
   425	    id_spec = f"featuring {primary_name}" + (f" ({', '.join(parts)})" if parts else "")
   426	    description = f"{id_spec}, {valid_bio}, styled in {request.get('tone', 'parody')} tone."
   427	
   428	    return {
   429	        "names": names,
   430	        "character": character_input,
   431	        "actor": actor_input,
   432	        "tone": request.get("tone", "parody"),
   433	        "length_s": request.get("length_s", 10),
   434	        "mode": mode,
   435	        "index_status": best_status,
   436	        "index_flag": "index:none" if best_status == "none" else f"index:{best_status}",
   437	        "index_rows": matched_index_rows,
   438	        "description": description,
   439	        "refmod_slots": slots,
   440	        "loaded_refmods": [str(m[0]) for m in kept_refmods],
   441	        "dropped_refmods": [str(m[0]) for m in dropped_refmods],
   442	        "stills": validated_stills,
   443	    }

   463	def _keep_highest_oc_version(stems: Sequence[str]) -> List[str]:
   464	    """When multiple versions (v1, v2) of the same OC key appear in stems, keep only the highest version.
   465	
   466	    A voice RefMod trimmed to fit a continuation beat's cond_audio cap (`<name>_23s`,
   467	    build_request.CONTINUATION_VOICE_REFMOD_SUFFIX) is a DERIVED copy of a reference that exists in full form:
   468	    build_request._continuation_voice_refmods wires it on one continuation beat from that beat's own layout.
   469	    It is never a character's voice reference for a base beat - two audio RefMods there pack
   470	    2*(1000+920)=3840 rows against the 2000 a voice reference carries, the over-cap family of the live c2_samp
   471	    failure (2026-11-15, [2000,32] vs [2130,32]). So a trim is dropped exactly when the file it was cut from is
   472	    in the same list; a list with only a trim keeps it (a character whose full reference is not installed).
   473	    """
   474	    max_ver: Dict[str, int] = {}
   475	    for s in stems:
   476	        m = _OC_KEY_RE.search(s)
   477	        if m:
   478	            vm = _OC_VER_RE.search(s)
   479	            ver = int(vm.group(1)) if vm else 0
   480	            k = m.group(1).lower()
   481	            if ver > max_ver.get(k, -1):
   482	                max_ver[k] = ver
   483	    out: List[str] = []
   484	    for s in stems:
   485	        m = _OC_KEY_RE.search(s)
   486	        if m:
   487	            vm = _OC_VER_RE.search(s)
   488	            ver = int(vm.group(1)) if vm else 0
   489	            k = m.group(1).lower()
   490	            if ver < max_ver.get(k, 0):
   491	                continue
   492	        out.append(s)
   493	    trim = _CONTINUATION_VOICE_SUFFIX
   494	    drop = {s for s in out if str(s).endswith(trim) and str(s)[: -len(trim)] in out}
   495	    return [s for s in out if s not in drop]

   560	def _safetensors_metadata(path: Path) -> Dict[str, Any]:
   561	    """Read the safetensors JSON header only (no tensor load)."""
   562	    try:
   563	        with path.open("rb") as f:
   564	            n = int.from_bytes(f.read(8), "little")
   565	            if n <= 0 or n > 4_000_000:
   566	                return {}
   567	            raw = f.read(n)
   568	        header = json.loads(raw.decode("utf-8"))
   569	        meta = header.get("__metadata__") or {}
   570	        if not isinstance(meta, dict):
   571	            return {}
   572	        out: Dict[str, Any] = {}
   573	        for k, v in meta.items():
   574	            if isinstance(v, str):
   575	                try:
   576	                    out[k] = json.loads(v)
   577	                except Exception:
   578	                    out[k] = v
   579	            else:
   580	                out[k] = v
   581	        return out
   582	    except Exception:
   583	        return {}

   625	def pixlstash_character_names_for_still(still_image: Optional[str]) -> List[str]:
   626	    """PixlStash character names tagged on this still. Empty if the vault is missing."""
   627	    if not still_image:
   628	        return []
   629	    rel = "/".join(_still_path_parts(still_image))
   630	    candidates = {_norm_fs(still_image)}
   631	    try:
   632	        resolved = Path(still_image)
   633	        if resolved.exists():
   634	            candidates.add(_norm_fs(str(resolved.resolve())))
   635	    except Exception:
   636	        pass
   637	    if rel:
   638	        for root in _STILL_LIBRARY_ROOTS:
   639	            candidates.add(_norm_fs(str(root / rel)))
   640	        candidates.add(_norm_fs(rel))
   641	    names: List[str] = []
   642	    import sqlite3
   643	    for vault in _PIXLSTASH_VAULTS:
   644	        if not vault.is_file():
   645	            continue
   646	        try:
   647	            con = sqlite3.connect("file:%s?mode=ro" % vault.as_posix(), uri=True)
   648	            cur = con.cursor()
   649	            rows = cur.execute(
   650	                """
   651	                SELECT c.name, p.file_path
   652	                FROM face f
   653	                JOIN character c ON c.id = f.character_id
   654	                JOIN picture p ON p.id = f.picture_id
   655	                WHERE f.character_id IS NOT NULL
   656	                """
   657	            ).fetchall()
   658	            con.close()
   659	        except Exception:
   660	            continue
   661	        for name, fpath in rows:
   662	            if not name:
   663	                continue
   664	            fp = _norm_fs(fpath)
   665	            if fp in candidates or (rel and (fp.endswith("/" + rel.lower()) or fp == rel.lower())):
   666	                if name not in names:
   667	                    names.append(name)
   668	        if names:
   669	            return names
   670	    return names

   748	def index_named_identity(name: Optional[str]) -> bool:
   749	    """True when the name is a character INDEX or aliases entry, not an installed OC key."""
   750	    if not (name or "").strip():
   751	        return False
   752	    n = _norm_ident(name)
   753	    if n in installed_oc_refmod_keys() or n in trained_oc_subject_keys():
   754	        return False
   755	    try:
   756	        aliases = load_aliases(ALIASES_PATH) or {}
   757	        for k, v in aliases.items():
   758	            if _name_matches_field(n, k) or _name_matches_field(n, v):
   759	                return True
   760	        for row in load_index_files(INDEX_DIR) or []:
   761	            if _name_matches_field(n, str(row.get("character") or "")) or _name_matches_field(n, str(row.get("actor") or "")):
   762	                return True
   763	    except Exception:
   764	        return False
   765	    return False

   768	def resolve_writer_refmods(
   769	    still_image: Optional[str] = None,
   770	    character: Optional[str] = None,
   771	    names: Optional[Sequence[str]] = None,
   772	) -> Dict[str, Any]:
   773	    """Pick installed visual/audio RefMods for the writer hook across OC, INDEX, and PixlStash tags.
   774	
   775	    Uses aliases + INDEX + local RefMod files (all adult RefMods and voice refs allowed), trained
   776	    RefMod triggers (e.g. FCSydny), PixlStash character tags, or approved POOL path components.
   777	    Never guesses a likeness from a bare still filename, and never reuses the previous run's RefMods.
   778	    Stills are not passed to validate_stills (writer stills may live outside approved_characters).
   779	    """
   780	    out: Dict[str, Any] = {"stems": [], "source": None, "reason": None, "index": None, "trigger": None}
   781	    char = (character or "").strip() or None
   782	    name_list = [n for n in (list(names) if names else []) if str(n).strip()]
   783	
   784	    def _oc_trigger(stems: Sequence[str]) -> Optional[str]:
   785	        """The trained RefMod trigger word (e.g. FCSydny) of the visual RefMod just attached, from its own
   786	        safetensors subject_name. Only for minimaxh3_oc_* files; None for INDEX/celebrity RefMods."""
   787	        d = Path(REFMODS_DIR)
   788	        for s in stems:
   789	            stem = _mod_stem(str(s))
   790	            if not re.search(r"(?i)^minimaxh3_oc_", stem):
   791	                continue
   792	            f = d / (stem + ".safetensors")
   793	            if not f.is_file():
   794	                continue
   795	            subj = _safetensors_metadata(f).get("subject_name")
   796	            if isinstance(subj, str) and subj.strip():
   797	                return subj.strip()
   798	        return None
   799	
   800	    def _resolve_names(targets: Sequence[str]) -> List[str]:
   801	        found: List[str] = []
   802	        seen: set = set()
   803	        for n in targets:
   804	            for s in _stems_for_character_name(str(n)):
   805	                if _is_writer_lane_refmod(s) and s not in seen:
   806	                    seen.add(s)
   807	                    found.append(s)
   808	        try:
   809	            tlist = [str(n).strip() for n in targets if str(n).strip()]
   810	            req = {
   811	                "names": tlist,
   812	                "character": tlist[0] if tlist else "",
   813	                "actor": "",
   814	                "tone": "",
   815	                "length_s": 10,
   816	                "description": "adult figure with eyes, hair, and outfit",
   817	            }
   818	            resolved = resolve_identities(req, index_dir=INDEX_DIR, refmods_dir=REFMODS_DIR, aliases_path=ALIASES_PATH)
   819	            loaded = [_mod_stem(x) for x in (resolved.get("loaded_refmods") or [])]
   820	            out["index"] = {
   821	                "index_status": resolved.get("index_status"),
   822	                "index_flag": resolved.get("index_flag"),
   823	                "loaded": loaded,
   824	            }
   825	            for s in loaded:
   826	                if _is_writer_lane_refmod(s) and s not in seen:
   827	                    seen.add(s)
   828	                    found.append(s)
   829	        except Exception as e:
   830	            out["index_error"] = f"{type(e).__name__}: {e}"
   831	        return _keep_highest_oc_version(found)
   832	
   833	    if char or name_list:
   834	        targets = ([char] if char else []) + [n for n in name_list if n != char]
   835	        matched = _resolve_names(targets)
   836	        if matched:
   837	            out["stems"] = matched
   838	            out["source"] = "character"
   839	            out["trigger"] = _oc_trigger(matched)
   840	            return out
   841	        who = char or (name_list[0] if name_list else "named identity")
   842	        out["reason"] = f"no local RefMod installed for '{who}' in {REFMODS_DIR}"
   843	        return out
   844	
   845	    keys = oc_keys_for_still_and_names(still_image, char, " ".join(str(n) for n in name_list))
   846	    if keys:
   847	        inst = installed_oc_refmod_keys()
   848	        stems: List[str] = []
   849	        seen: set = set()
   850	        for k in keys:
   851	            for s in inst.get(k, []):
   852	                if s not in seen:
   853	                    seen.add(s)
   854	                    stems.append(s)
   855	            if not inst.get(k):
   856	                for s in _stems_for_character_name(k):
   857	                    if _is_writer_lane_refmod(s) and s not in seen:
   858	                        seen.add(s)
   859	                        stems.append(s)
   860	        if stems:
   861	            out["stems"] = _keep_highest_oc_version(stems)
   862	            out["source"] = "still_path"
   863	            out["trigger"] = _oc_trigger(out["stems"])
   864	            return out
   865	
   866	    tagged = pixlstash_character_names_for_still(still_image)
   867	    if tagged:
   868	        matched = _resolve_names(tagged)
   869	        if matched:
   870	            out["stems"] = matched
   871	            out["source"] = "pixlstash"
   872	            out["trigger"] = _oc_trigger(matched)
   873	            return out
   874	
   875	    out["reason"] = (
   876	        "no RefMod matched this still via installed files, the trained trigger, "
   877	        "the POOL folder, a PixlStash character tag, or the character name; "
   878	        "will not reuse the previous run's RefMods. Pass a still tagged or filed as the character "
   879	        "(for example FCSydny / poolclub or poolblonde), or pass --character <name>."
   880	    )
   881	    return out
