   102	def pick_tone(ask: str, rng: random.Random, last: Optional[str], overlays: Optional[List[str]] = None) -> str:
   103	    """Reluctant/ashamed only when the ask or a cnc/defeat/forced overlay names it (PM acc4 #15)."""
   104	    for tone, pat in TONE_WORDS:
   105	        if re.search(r"(?i)\b(?:" + pat + r")", ask or ""):
   106	            return tone
   107	    overlays = overlays or []
   108	    named = any(o in overlays for o in ("cnc", "defeat", "forced_orgasm"))
   109	    choices = [t for t in TONES if t != last]
   110	    if not named:
   111	        choices = [t for t in choices if t not in ("ashamed", "reluctant")]
   112	    if not choices:
   113	        choices = ["eager", "bratty", "silent"]
   114	    return rng.choice(choices)

   126	def normalize_ending_state(es: Optional[Dict[str, Any]]) -> Dict[str, Any]:
   127	    """Clean inherited end state before a continued opener (PM acc4d #4): never 'sheathed' in the mouth."""
   128	    out = dict(es or {})
   129	    for k in ("pose", "contact"):
   130	        v = out.get(k)
   131	        if not v:
   132	            continue
   133	        s = str(v)
   134	        s = _SHEATHED_MOUTH_RE.sub("inside her mouth", s)
   135	        s = re.sub(r"(?i)\bsheathed\b(?=[^.;]{0,40}\b(?:mouth|throat|lips)\b)", "inside", s)
   136	        if _NO_CONTACT_RE.match(s.strip()):
   137	            s = ""
   138	        out[k] = s
   139	    return out

   202	def section_split(text: str) -> List[Tuple[str, List[str]]]:
   203	    out: List[Tuple[str, List[str]]] = []
   204	    for ln in text.splitlines():
   205	        if re.match(r"^[A-Z][A-Z ,&/-]{3,}(?:\s*\(.*\))?$", ln.strip()) and not ln.startswith("-"):
   206	            out.append((ln.strip(), []))
   207	        elif out:
   208	            out[-1][1].append(ln)
   209	        else:
   210	            out.append(("", [ln]))
   211	    return out

   214	def assemble_system(cfg: Dict[str, Any], ask: str, persona: str, blocks: Optional[List[str]] = None) -> Tuple[str, Dict[str, Any]]:
   215	    """writer_prompt.md with the chosen persona and only the mode block(s) the ask names (spec section 4 item 10)."""
   216	    wp = Path(cfg.get("writer_prompt") or "writer_prompt.md")
   217	    if not wp.is_absolute():
   218	        wp = PIPE_DIR / wp
   219	    text, meta = build_writer_system_prompt([wp])  # refuses AGENTS.md names and agent-only markers
   220	    text = text.split("=====\n", 1)[-1]
   221	    parts = section_split(text)
   222	    out = []
   223	    modes_used = []
   224	    for head, lines in parts:
   225	        if head == "PERSONA" and persona and persona != "current":
   226	            pfile = PIPE_DIR / "personas" / f"{persona}.md"
   227	            lines = ["", pfile.read_text(encoding="utf-8").strip(), ""]
   228	        if head.startswith("MODES"):
   229	            keep = []
   230	            cur_ok = False
   231	            for ln in lines:
   232	                if ln.startswith("- "):
   233	                    cur_ok = False
   234	                    for label, pat in MODE_TRIGGERS:
   235	                        hit = (label[2:] in blocks) if blocks is not None else bool(re.search(r"(?i)" + pat, ask or ""))
   236	                        if ln.startswith(label) and hit:
   237	                            cur_ok = True
   238	                            modes_used.append(label[2:])
   239	                if cur_ok or not ln.strip():
   240	                    keep.append(ln)
   241	            if not modes_used:
   242	                continue
   243	            lines = keep
   244	        out.append(((head + "\n") if head else "") + "\n".join(lines).strip("\n"))
   245	    sysprompt = "\n\n".join(o for o in out if o.strip())
   246	    meta = {"writer_prompt": str(wp), "writer_prompt_sha256": wp_state.sha256_file(wp), "persona": persona or "current",
   247	            "modes": modes_used}
   248	    return sysprompt, meta

   452	def schema_errors(obj: Any, schema: Dict[str, Any], path: str = "$") -> List[str]:
   453	    errs: List[str] = []
   454	    t = schema.get("type")
   455	    types = {"object": dict, "array": list, "string": str, "integer": int, "number": (int, float), "boolean": bool}
   456	    allowed = [t] if isinstance(t, str) else (list(t) if isinstance(t, list) else [])
   457	    if allowed:
   458	        ok = any(a in types and isinstance(obj, types[a]) and not (a in ("integer", "number") and isinstance(obj, bool))
   459	                 for a in allowed)
   460	        if not ok:
   461	            return [f"{path}: expected {t}"]
   462	    if isinstance(obj, dict) and (t == "object" or "object" in allowed):
   463	        for k in schema.get("required", []):
   464	            if k not in obj:
   465	                errs.append(f"{path}.{k}: missing")
   466	        for k, sub in (schema.get("properties") or {}).items():
   467	            if k in obj:
   468	                errs += schema_errors(obj[k], sub, f"{path}.{k}")
   469	    if isinstance(obj, list) and (t == "array" or "array" in allowed):
   470	        if len(obj) < schema.get("minItems", 0) or len(obj) > schema.get("maxItems", 10 ** 6):
   471	            errs.append(f"{path}: {len(obj)} items")
   472	        for i, it in enumerate(obj):
   473	            errs += schema_errors(it, schema.get("items") or {}, f"{path}[{i}]")
   474	    if isinstance(obj, str) and not obj.strip():
   475	        leaf = path.rsplit(".", 1)[-1]
   476	        if leaf not in ("contact", "hair", "species", "scale_anchor", "anatomy", "matches_still", "still", "prop_hand"):
   477	            errs.append(f"{path}: empty")
   478	    return errs

   812	def normalize_prose(dd: str, kind: str, slots: Optional[Dict[str, Any]] = None) -> str:
   813	    """Mechanical fixes only (no new content): unwrap <d>[words]</d> brackets, and put the literal 'PRIMARY MOTION - '
   814	    label on the 'As it builds' sentence when the writer wrote the motion with a tempo word but left the label off."""
   815	    def _d(m: "re.Match[str]") -> str:
   816	        inner = m.group(1).strip()
   817	        if not _LANG_TAG_RE.match(inner):
   818	            mm = re.fullmatch(r"\[(.+)\]", inner)
   819	            if mm:
   820	                inner = mm.group(1).strip()
   821	        return f"<d>{inner}</d>"
   822	    # builder _HIS_CLOTHES_RE (build_request.py:795) matches 'stays bunched' and drops the whole clause, which deletes the
   823	    # clearance and trips W5 (final4_c missionary/doggy). Same meaning, a word the builder keeps:
   824	    dd = re.sub(r"(?i)\b(stay(?:s|ing)?|left|remain(?:s|ing)?) bunched\b", r"\1 gathered", dd)
   825	    # builder _HIS_LIFT_RE (build_request.py:2233) wants 'his hips ... lift' (m09): same meaning, builder word order
   826	    dd = re.sub(r"\((S2)\) lifts his (hips|pelvis)\b", r"his \2 lift", dd)
   827	    dd = re.sub(r"(?i)\bhe lifts his (hips|pelvis)\b", r"his \1 lift", dd)
   828	    # dry-matte skin (Derrick floor): shiny skin words from the read or the writer -> damp
   829	    dd = re.sub(r"(?i)\b(?:glistening|glossy|oily|oiled|shimmering)\b", "damp", dd)
   830	    dd = re.sub(r"(?i)\bdamp,? damp\b", "damp", dd)
   831	    # act ids never ship as prose (c8b 'At first, standing_mf;'): builder ids -> plain words
   832	    surf = (slots or {}).get("surface") or "surface"
   833	    for aid, words in (("standing_mf", f"standing, bent forward over the {surf}"), ("reverse_cowgirl", "reverse cowgirl"),
   834	                       ("mf_anal", "anal"), ("female_solo", "alone"), ("standing_ff", "standing")):
   835	        dd = re.sub(r"\b" + aid + r"\b", words, dd)
   836	    # strip/solo tempo 'every two seconds' is not a builder tempo word (_TEMPO_RE): same tempo, with 'cycle'
   837	    if wp_lint.mode_rules(kind)["primary"] and not br._TEMPO_RE.search(re.sub(r"<d>.*?</d>", "", dd, flags=re.S)):
   838	        dd = re.sub(r"(?i)\b(?:about\s+)?(?:one\s+)?(?:slow\s+|smooth\s+|steady\s+)?(?:move|pull|tug|motion|stroke|peel|step)?\s*"
   839	                    r"every (two|three|four|five|2|3|4|5) seconds\b", r"one slow cycle every \1 seconds", dd, count=1)
   840	    # acceptance-3 (m18): 'two thrusts a second' / 'pumps a second' -> the builder tempo word 'strokes a second'
   841	    dd = re.sub(r"(?i)\b(thrusts?|pumps?|bobs?|pushes|drives|slams?|bounces?)( a second)\b", "strokes a second", dd)
   842	    # tag words never ship (Derrick floor): 'cinematic'/'beautiful' -> drop the adjective
   843	    dd = re.sub(r"(?i)\s*\b(?:cinematic|beautiful(?:ly)?)\b,?", "", dd)
   844	    for k in ("place", "light", "surface", "hair"):  # pmfix6: SHAPE blanks left literal -> fill from still_read slots
   845	        if (slots or {}).get(k):
   846	            dd = re.sub(r"<\s*" + k + r"\s*>", str(slots[k]), dd, flags=re.I)
   847	    for _ in range(3):  # unwrap nested <d><d>..</d></d> the writer sometimes produces
   848	        dd = re.sub(r"<d>\s*(?:\[English\]\s*)?<d>(.*?)</d>\s*</d>", r"<d>\1</d>", dd, flags=re.S)
   849	    dd = re.sub(r"<d>(.*?)</d>", _d, dd, flags=re.S)
   850	    dd = re.sub(r"<d>(\s*(?:\[English\]\s*)?)[\"\u201c\u201d']+(.*?)[\"\u201c\u201d']+</d>", r"<d>\1\2</d>", dd, flags=re.S)
   851	    dd = re.sub(r"(?<![(\w<])S([123])(?![\w)>])", r"(S\1)", dd)  # bare S1 -> (S1)

   853	    _mk = re.sub(r"<d>.*?</d>", lambda mm: "\x00" * len(mm.group(0)), dd, flags=re.S)
   854	    _mc = wp_lint.MOUTH_CONTACT_RE.search(_mk)
   855	    if _mc:
   856	        head, tail = dd[:_mc.end()], dd[_mc.end():]
   857	        tail = re.sub(r"(?:(?<=[.!?>])|^)\s*\(S1\)[^<.!?]{0,120}<d>.*?</d>[.!?]?", "", tail, flags=re.S)
   858	        dd = head + tail
   859	    if wp_lint.mode_rules(kind)["primary"] and not br._PRIMARY_RE.search(dd):
   860	        # the first sentence (outside <d>) that carries a tempo word gets the label, after a leading 'As it builds,'
   861	        # acceptance-2 bug 1: mask every <d>...</d> span first, so a line ending in '?' or '!' never puts the label
   862	        # inside the dialogue; a chunk that holds a masked span is only searched after its last span
   863	        masked = re.sub(r"<d>.*?</d>", lambda mm: "\x00" * len(mm.group(0)), dd, flags=re.S)
   864	        for m in re.finditer(r"[^.!?]*[.!?]", masked):
   865	            sent = m.group(0)
   866	            off = sent.rfind("\x00") + 1

   871	            at = m.start() + off + lead.end()
   872	            dd = dd[:at] + "PRIMARY MOTION - " + dd[at:]
   873	            break
   874	    return glue_fix(dd)

  1077	def judge_letter(jd: Dict[str, Any], letters: List[str]) -> str:
  1078	    """'CANDIDATE A', 'A', 'Candidate: B', ranking[0] or the top score -> one letter that exists (fixes 'CANDIDATE A' -> 'C')."""
  1079	    def one(v: Any) -> str:
  1080	        t = str(v or "").strip().upper()
  1081	        m = re.fullmatch(r"(?:CANDIDATE\s*[:#-]?\s*)?([A-Z])\.?", t) or re.search(r"\b([A-Z])\b(?!.*\b[A-Z]\b)", t)
  1082	        return m.group(1) if m and m.group(1) in letters else ""
  1083	    b = one(jd.get("best"))
  1084	    if not b and isinstance(jd.get("ranking"), list) and jd["ranking"]:
  1085	        b = one(jd["ranking"][0])
  1086	    if not b and isinstance(jd.get("scores"), dict):
  1087	        tot = lambda v: v if isinstance(v, (int, float)) else (v.get("total") if isinstance(v, dict) and isinstance(v.get("total"), (int, float))
  1088	                                                                else sum(x for x in v.values() if isinstance(x, (int, float))) if isinstance(v, dict) else None)
  1089	        sc = {one(k): tot(v) for k, v in jd["scores"].items() if one(k) and tot(v) is not None}

  1344	def _file_sha256(path: Path) -> Optional[str]:
  1345	    try:
  1346	        h = hashlib.sha256()
  1347	        with open(path, "rb") as f:
  1348	            for chunk in iter(lambda: f.read(1 << 20), b""):
  1349	                h.update(chunk)
  1350	        return h.hexdigest()
  1351	    except OSError:
  1352	        return None

  1359	def still_read_mismatch(still: str, run_dir: Path, still_read_src: Optional[str]) -> Tuple[Optional[str], Optional[str]]:
  1360	    """(not_ready_reason, warning) for the still read the pipeline is about to use (run_dir/still_read.json if present,
  1361	    else --still-read). It must belong to the start still: same path, or the same file bytes (sha256). A read with no
  1362	    "source" (and no "source_sha256") cannot be checked and only warns."""
  1363	    existing = run_dir / "still_read.json"
  1364	    srp = existing if existing.exists() else (Path(still_read_src) if still_read_src else None)
  1365	    if srp is None:
  1366	        return None, None
  1367	    try:
  1368	        d = json.loads(srp.read_text(encoding="utf-8"))
  1369	    except Exception as e:  # noqa: BLE001
  1370	        return f"still read {srp} is not valid JSON ({e})", None
  1371	    d = d if isinstance(d, dict) else {}
  1372	    sha = str(d.get("source_sha256") or "").strip().lower()
  1373	    src = d.get("source")
  1374	    if sha:
  1375	        have = _file_sha256(Path(still))
  1376	        if have and have == sha:
  1377	            return None, None
  1378	        return (f"still read does not belong to the start still: {srp} source_sha256 does not match --still '{still}'"), None
  1379	    if not src:
  1380	        return None, "still_read_unverified (no source in the still read)"
  1381	    if _norm_still(src) == _norm_still(still):
  1382	        return None, None
  1383	    try:
  1384	        if os.path.realpath(os.path.expanduser(str(src))) == os.path.realpath(os.path.expanduser(str(still))):
  1385	            return None, None
  1386	    except Exception:  # noqa: BLE001
  1387	        pass
  1388	    a, b = _file_sha256(Path(os.path.expanduser(str(src)))), _file_sha256(Path(still))
  1389	    if a and b and a == b:
  1390	        return None, None
  1391	    return (f"still read does not belong to the start still: {srp.name} describes '{src}' but --still is '{still}'; "
  1392	            "pass the still that read describes, or remove the read so the pipeline reads the new still"), None
