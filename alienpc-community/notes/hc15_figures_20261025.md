## 5. Reproduce (no new script file; run in-place)

```
cd C:/AI/community
python - <<'EOF'   # bytes + colour fractions of every LoadImage file in the six HC15 graphs
import importlib.util, os, hashlib, numpy as np
from PIL import Image
s = importlib.util.spec_from_file_location("cb", "notes/hc15_anchor_carry_build.py")
cb = importlib.util.module_from_spec(s); s.loader.exec_module(cb)
ids = [v[1] for v in cb.CARRY.values()] + ["ali163_hc15_gi046_s51150014", "ali163_hc15_gi063_s51150015",
                                           "ali163_hc15_bd002_s51150011"]
for nid in set(ids):
    for n in cb.gload(nid).values():
        if n.get("class_type") == "LoadImage":
            p = os.path.join(cb.COMFY_IN, n["inputs"]["image"].replace("/", os.sep))
            a = np.asarray(Image.open(p).convert("RGB")).astype(int)
            r, g, b = a[..., 0], a[..., 1], a[..., 2]
            mx, mn = a.max(2), a.min(2); sat = (mx - mn) / np.maximum(mx, 1)
            print(nid, n["inputs"]["image"], Image.open(p).size, os.path.getsize(p),
                  hashlib.sha256(open(p, "rb").read()).hexdigest(),
                  round(float(((g > r + 8) & (g > b + 8)).mean()), 4),
                  round(float(((sat < 0.2) & (mx > 60)).mean()), 4))
EOF
python notes/hc15_anchor_carry_build.py --top-check     # the prose/graph facts above, re-derived off disk
```

Colour probe thresholds are stated in §2 so a re-run reproduces the numbers; they are a support test,
not a species classifier, and no species claim in this file rests on them alone.
