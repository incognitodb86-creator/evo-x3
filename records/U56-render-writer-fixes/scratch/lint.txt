    93	def tempo_rate(text: str) -> Optional[float]:
    94	    m = PER_SEC_RE.search(text or "")
    95	    if not m:
    96	        return None
    97	    n = _NUM.get(m.group(1).lower()) or float(m.group(1))
    98	    return n * (2 if m.group(2) else 1)

   101	def speaker_counts(dd: str) -> Dict[str, int]:
   102	    out: Dict[str, int] = {}
   103	    text = dd or ""
   104	    for m in SPEAKER_D_RE.finditer(text):
   105	        pre = text[max(0, m.start() - 100):max(0, m.end() - 3)]
   106	        spk = "?" if _INTERVIEWER_LEAD_RE.search(pre) else (m.group(1) or "?")
   107	        out[spk] = out.get(spk, 0) + 1
   108	    return out

   111	def oral_mouth_talk(dd: str) -> bool:
   112	    """True when (S1) has a <d> line while her mouth is on him (the builder's D1 rule, imported regexes)."""
   113	    masked = D_RE.sub(lambda x: " " * len(x.group(0)), dd)
   114	    ev = sorted([(m.start(), "on") for m in ORAL_ACT_RE.finditer(masked)] + [(m.start(), "off") for m in MOUTH_FREE_RE.finditer(masked)]
   115	                + [(m.start(), "d") for m in SPEAKER_D_RE.finditer(dd) if (m.group(1) or "") == "S1"])
   116	    on = False
   117	    for _, k in ev:
   118	        if k == "on":
   119	            on = True
   120	        elif k == "off":
   121	            on = False
   122	        elif on:
   123	            return True
   124	    return False

   127	def mode_checks(dd: str, full: str, sents: List[str], plain: str, ctx: Dict[str, Any], kind: str, prim_idx: Optional[int]) -> List[Dict[str, Any]]:
   128	    """Builder-parity and overlay rules (coverage addendum B3); hard unless marked soft."""
   129	    out: List[Dict[str, Any]] = []
   130	    cast = ctx.get("cast") or {}
   131	    ov = set(ctx.get("overlays") or [])
   132	    ask = ctx.get("ask") or ""
   133	    act = ctx.get("act_id") or ""
   134	    cfg = ctx.get("cfg") or {}
   135	    # L-POS: the act's landing words (builder position_check) in the prose
   136	    pc = cast.get("position_check")
   137	    if pc and kind != "talk" and not re.search(pc, plain, re.I):
   138	        out.append(_fix("pose_landing", f"land the pose in the builder's words for {act} (e.g. one of: {pc[:120]})"))
   139	    # L-PARTNER: an off-still partner must be defined (builder writer_defined_missing_person)
   140	    if cast.get("off_still_s2") and not br.writer_defined_missing_person(full, 2):
   141	        out.append(_fix("partner_gate", "define (S2) when he enters: look or clothes and 'enters from' in one sentence"))
   142	    if cast.get("off_still_s3") and not br.writer_defined_missing_person(full, 3):
   143	        out.append(_fix("partner_gate_s3", "define (S3) when they enter: look or clothes and 'enters from' in one sentence"))
   144	    # L-VOICE (acceptance-1 f): a speaker the builder has no voice for gets 'a distinct adult voice for (Sn)'
   145	    if cast:
   146	        for n in sorted(set(re.findall(r"\(S([23])\)[^<.!?]{0,120}<d>", dd))):
   147	            if not cast.get(f"voice_s{n}"):
   148	                if re.search(rf"(?i)\(S{n}\)[^<.!?]{{0,60}}\boff[- ]?screen", dd):
   149	                    out.append(_fix("voice_missing", f"off-screen interviewer has no (S{n}) label; rewrite without (S{n}) as 'An off-screen interviewer asks, <d>...</d>'"))
   150	                else:
   151	                    out.append(_fix("voice_missing", f"(S{n}) has no voice in this clip's cast (talk/solo act); give the line to (S1) "
   152	                                                     f"or make it an off-screen voice"))
   153	    # L-FUTA: builder P2-FUTA-AS-MAN
   154	    if re.search(r"(?i)\bfuta\b", ask) and re.search(r"(?i)\b(?:he|his|him|himself|the man)\b", plain):
   155	        out.append(_fix("futa_pronoun", "the futa partner is she/her; no he, his, him or the man"))
   156	    # L-SPK2: the builder keeps at most max_lines_per_speaker lines per speaker
   157	    max_spk = int(cfg.get("max_lines_per_speaker", 2))
   158	    for spk, n in speaker_counts(dd).items():
   159	        if n > max_spk:
   160	            out.append(_fix("speaker_cap", f"({spk}) has {n} lines; the builder keeps {max_spk} per speaker, so give at most {max_spk}"))
   161	    # acceptance-2 item 10: silent mouth in every act (oral, mmf/spitroast, finish, titjob) once her mouth is on him
   162	    first_contact = next((i for i, x in enumerate(sents) if MOUTH_CONTACT_RE.search(outside_d(x))), None)
   163	    if first_contact is not None and any(re.search(r"\(S1\)[^<]{0,120}<d>", x) for x in sents[first_contact:]):
   164	        out.append(_fix("oral_silence", "after her mouth is on him only (S2) speaks; give (S1) wet suction and breath instead"))
   165	    if (kind == "oral" or act in ("finish", "titjob", "blowjob") or "toy" in ov) and SLAP_RE.search(plain):
   166	        out.append(_fix("oral_sound", "no slap foley here: wet suction, breath, fabric or the toy's hum"))
   167	    if re.search(r"<d>[^<]*<d>|</d>\s*</d>", dd) or any("[" in re.sub(r"^\s*\[English\]\s*", "", m) or "PRIMARY MOTION" in m
   168	                                                       for m in re.findall(r"<d>(.*?)</d>", dd, flags=re.S)):
   169	        out.append(_fix("nested_d", "one <d>[English] words</d> pair per spoken line: no nesting, no other [brackets], no labels inside"))
   170	    if TAG_WORDS_RE.search(full):
   171	        out.append(_fix("tag_words", "remove the words cinematic / beautiful (summary and prose)"))
   172	    # act_mismatch: another act's landing words in a non-transition clip (c3b standing_mf landing missionary)
   173	    if kind not in ("transition", "talk") and act in ACT_LANDING:
   174	        body_s = [outside_d(x) for x in sents if not OPENER_RE.search(x)]
   175	        for other, rx in ACT_LANDING.items():
   176	            if other != act and other not in ACT_COMPAT.get(act, ()) and any(re.search(rx, x) for x in body_s):
   177	                out.append(_fix("act_mismatch", f"this clip is {act}; remove the {other} landing words and land in {act}'s own pose"))
   178	                break
   179	    if act == "standing_mf" and any(re.search(r"(?i)hands and knees|on all fours|(?<!standing )\bdoggy\b", outside_d(x))
   180	                                    for x in sents if not OPENER_RE.search(x)):  # PM acc3 #4
   181	        out.append(_fix("act_mismatch", "standing_mf lands 'bent forward over the surface, both feet planted'; never hands and knees, all fours or doggy"))
   182	    _pt = ctx.get("s3_picture") or 2  # PM acc3 #8: a <Picture N> tag (N>=2) only next to (S3)/<Subject 3>
   183	    for x in sents:
   184	        if re.search(rf"<Picture {_pt}>", x) and not re.search(r"\(S3\)|<Subject 3>", x):
   185	            out.append(_fix("picture_tag", f"<Picture {_pt}> belongs only in a sentence about (S3)"))
   186	            break
   187	    if kind == "oral" and oral_mouth_talk(dd):
   188	        out.append(_fix("oral_mouth_full", "(S1) speaks only while her mouth is free; move her line before he enters her mouth or after she pulls off"))
   189	    pm_s = outside_d(sents[prim_idx]) if prim_idx is not None else ""
   190	    if act == "cowgirl" and not (RISE_RE.search(pm_s) and SINK_RE.search(pm_s) and HIS_LIFT_RE.search(pm_s)
   191	                                 and br._LIFT_MEASURE_RE.search(pm_s) and MEET_RE.search(pm_s)):
   192	        out.append(_fix("ride_both", "write this clause word for word in the PRIMARY MOTION sentence: 'she rises and sinks on him, his hips "
   193	                                     f"lifting a few inches off the {(ctx.get('slots') or {}).get('surface') or 'surface'} to meet her as she sinks'"))
   194	    if DIAL_RE.search(plain) or re.search(r"(?i)\b\d{1,3}fps\b", plain):
   195	        _dm = DIAL_RE.search(plain) or re.search(r"(?i)\b\d{1,3}fps\b", plain)
   196	        out.append(_fix("dial", f"remove the technical term '{_dm.group(0)}'; film look is light, lens and grain words"))
   197	    if THROAT_RE.search(plain):
   198	        out.append(_fix("throat", "write throat contact exactly as 'his hand on her throat' (never slide/grip/close/choke throat wording)"))
   199	    if kind == "sex" and SHALLOW_RE.search(plain):  # partner (titjob/finish) has no entry depth
   200	        out.append(_fix("shallow", "the builder keeps full entry; show denial with tempo or a full pull-out pause"))
   201	    if NARR_RE.search(plain):
   202	        out.append(_fix("narrate", "show the dynamic in grip, gaze and breath instead of naming it"))
   203	    if PANEL_RE.search(plain):
   204	        out.append(_fix("panel", "name the garment by the still's own noun, never 'panel'"))
   205	    cam = [s for s in sents if re.search(r"(?i)\bcamera\b", s) and CAMERA_MOTION_RE.search(outside_d(s))]
   206	    if len(cam) > 1:
   207	        out.append(_fix("camera_jobs", f"one camera job per clip; {len(cam)} camera-move sentences"))
   208	    big = CAM_BIG_RE.search(" ".join(s for s in sents if re.search(r"(?i)\bcamera\b", s)))
   209	    if big and not CAM_BIG_RE.search(ask):
   210	        out.append(_fix("camera_big", f"use a stable camera move (held, slow push-in, slow pull-out); drop '{big.group(0)}'"))
   211	    fillers = [norm(f) for f in (cfg.get("filler_lines") or [])]
   212	    for l in spoken_lines(dd):
   213	        if any(f and re.search(r"(?<!\w)" + re.escape(f) + r"(?!\w)", l) for f in fillers) and l not in norm(ask):  # PM acc3 #8: substrings
   214	            out.append(_fix("filler", f"line '{l}' is a stock filler; write a line about the still"))
   215	    if "pov" in ov:
   216	        if re.search(r"(?i)\bhand-?held\b", " ".join(cam)) or re.search(r"(?i)\bextreme close", plain):
   217	            out.append(_fix("pov_camera", "POV: eye line of the dominant, medium or medium-close, one simple move, no handheld lead"))
   218	    if "restraints" in ov:
   219	        if not RESTRAINT_RE.search(plain):
   220	            out.append(_fix("restraint_named", "name the restraint, who set it and its limit once"))
   221	        if RELEASE_RE.search(plain) and not RELEASE_RE.search(ask):
   222	            out.append(_fix("restraint_release", "the restraint stays on to the last frame"))
   223	    if "cnc" in ov and SAFEWORD_RE.search(dd) and not SAFEWORD_RE.search(ask):
   224	        out.append(_fix("safeword", "no safeword unless the ask names one"))
   225	    if "taboo_role" in ov:
   226	        titles = {m.group(0).lower().replace("-", "") for m in ROLE_RE.finditer(dd)}
   227	        if len(titles) > 1:
   228	            out.append(_fix("role_titles", f"one role per clip; titles used: {', '.join(sorted(titles))}"))
   229	    rules = mode_rules(kind)
   230	    if rules.get("removals"):
   231	        n = len([s for s in sents if REMOVE_VERB_RE.search(outside_d(s))])
   232	        lo, hi = rules["removals"]
   233	        if not (lo <= n <= hi):
   234	            out.append(_fix("strip_removals", f"a strip clip has {lo}-{hi} removal steps, each with where the garment lands; found {n}"))
   235	        if CONTACT_RE.search(plain):
   236	            out.append(_fix("strip_contact", "a strip clip has no penetration contact"))
   237	    op = opening_sentence(dd)
   238	    if not (ctx.get("slots") or {}).get("inherited") and re.search(r"(?i)\bat first,\s*\(S1\)\s+(?:leans|sits|stands|is (?:leaning|sitting|standing))\b", dd):
   239	        out.append(_fix("opening_stock", "open 'At first,' on the lead detail from the brief, not '(S1) leans/sits/stands'", False, 5))
   240	    for prev in ctx.get("recent_openings") or []:
   241	        if op and prev and difflib.SequenceMatcher(None, op, prev).ratio() >= float(cfg.get("opening_ratio", 0.85)):
   242	            out.append(_fix("opening_repeat", "the 'At first' sentence repeats a recent clip; open on a different detail of the still"))
   243	            break
   244	    # soft
   245	    if (ctx.get("slots") or {}).get("surface_height") == "waist" and re.search(r"(?i)\b(?:he|\(S2\))\b[^.]{0,40}\bkneel\w*", plain):
   246	        out.append(_fix("height", "the surface is waist-high: he stands at its edge instead of kneeling on it", False, 10))
   247	    if WET_RE.search(plain) and not (ctx.get("wet_ok")):
   248	        out.append(_fix("wet_unasked", "wet, oil or sweat words only when the still or the ask has them", False, 5))
   249	    rate = tempo_rate(sents[prim_idx]) if prim_idx is not None else None
   250	    tr = (cfg.get("tempo_range") or {}).get("edge" if "edge" in ov else kind)
   251	    if rate is not None and tr and not (tr[0] <= rate <= tr[1]):
   252	        out.append(_fix("tempo_range", f"tempo {rate:g}/s is outside {tr[0]}-{tr[1]}/s for this mode", False, 5))
   253	    return out

   303	def split_sentences(dd: str) -> List[str]:
   304	    """Sentences of detailed_description, with <d> spans kept inside their sentence."""
   305	    body = re.sub(r"^\s*\[Shot 1\]\s*", "", (dd or "").strip())
   306	    spans: List[str] = []
   307	
   308	    def keep(m):
   309	        spans.append(m.group(0))
   310	        return f"\x00{len(spans) - 1}\x00"
   311	    masked = D_RE.sub(keep, body)
   312	    parts = [p.strip() for p in re.split(r"(?:(?<=[.!?])|(?<=\x00))\s+(?=[A-Z(<\x00\"])", masked) if p.strip()]
   313	    out = []

   319	        if merged and s.lstrip().lower().startswith("<d>"):

   321	        else:
   322	            merged.append(s)
   323	    return merged

   437	def blacklist_hits(lines: Sequence[str], blacklist: Iterable[str], ratio: float = 0.8) -> List[Dict[str, Any]]:
   438	    hits = []
   439	    bl = [norm(b) for b in blacklist if norm(b)]
   440	    for l in lines:
   441	        nl = norm(l)
   442	        for b in bl:
   443	            r = 1.0 if nl == b else difflib.SequenceMatcher(None, nl, b).ratio()
   444	            lw, bw = set(nl.split()) - _BL_STOP, set(b.split()) - _BL_STOP
   445	            jac = len(lw & bw) / max(1, len(lw | bw)) if lw and bw else 0.0
   446	            if r >= ratio or (jac >= 0.6 and len(lw & bw) >= 2):
   447	                hits.append({"line": l, "blacklisted": b, "ratio": round(r, 3)})
   448	                break
   449	    return hits

   661	
   662	    if _clear_needed(slots, kind) and not _join_inherited_clear(slots, ctx.get("scene")):

   668	        # clearance_missing (acceptance-2 c1a): the first entry/contact sentence comes before any clearance sentence

   706	    elif res is None:
   707	        issues.append(_fix("phase_result", "close the action with a result sentence ('by the end ...') where the motion is still going"))

   709	    if pure_action:

   711	        if not (ACT_RE.search(last) or CONTINUE_RE.search(last)):
   712	            issues.append(_fix("static_end", "the last action sentence must keep motion alive (a continuing -ing verb, 'keeps', 'still')"))
